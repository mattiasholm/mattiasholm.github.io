import useGlobalData from '@docusaurus/useGlobalData';
import React from 'react';
import { useHistory, useLocation } from '@docusaurus/router';

export function KeyFilter({ category }) {
  const stats = useGlobalData()['tunes-stats']?.default;
  const history = useHistory();
  const location = useLocation();
  const selectedKey = new URLSearchParams(location.search).get('key') ?? 'All';
  const setSelectedKey = (key) => {
    const params = new URLSearchParams(location.search);
    if (key === 'All') {
      params.delete('key');
    } else {
      params.set('key', key);
    }
    history.push({ ...location, search: params.toString() });
  };

  if (!stats) {
    return null;
  }

  const tunes = stats.tunesByCategory?.[category] ?? [];
  const keys = ['All', ...Array.from(new Set(tunes.map((tune) => tune.keySignature))).sort()];
  const filtered = selectedKey === 'All' ? tunes : tunes.filter((tune) => tune.keySignature === selectedKey);

  return (
    <div>
      <div>
        {keys.map((key) => (
          <button
            key={key}
            onClick={() => setSelectedKey(key)}
            className={selectedKey === key ? 'keyButton keyButton--selected' : 'keyButton'}
          >
            {key}
          </button>
        ))}
      </div>
      <ul>
        {filtered.map(({ title, url }) => (
          <li key={url}>
            <a href={url}>{title}</a>
          </li>
        ))}
      </ul>
    </div>
  );
}

<KeyFilter category='{{CATEGORY_SLUG}}' />
